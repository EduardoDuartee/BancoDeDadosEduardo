==========Criação de tabela==========

CREATE TABLE logistica(
	id INT AUTO_INCREMENT PRIMARY KEY,
	nome VARCHAR(100) NOT NULL,
	valor FLOAT NOT NULL,
	quantidade INT NOT NULL
);

========== Fim da Criação de tabela==========

==========Alterando nome com indereçamento id==========
