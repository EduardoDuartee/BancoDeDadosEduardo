==========Formas de Buscas==========

(=)SELECT * FROM produtos WHERE nome  = 'linux'; 
(>=)SELECT * FROM produtos WHERE preço  >= '99'; 
(<=)SELECT * FROM produtos WHERE preço  >= '99'; 
(!=)SELECT * FROM produtos WHERE preço  != '99'; 
(LIKE)SELECT * FROM produtos WHERE nome LIKE  'l%(começa), %l(termina) %l%'(possui no meio), 'l__as'(l+2 caracteres +as); 
(BETWEEN)SELECT * FROM produtos WHERE preço BETWEEN 99 AND 100;
(or)SELECT * FROM produtos WHERE categorias_id  = 1 or preço = 99;
(AND)SELECT * FROM produtos WHERE categorias_id = 1 AND preço >= 99;
